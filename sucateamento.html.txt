  function pl(l){const f=[];let c='',inQ=false;for(let i=0;i<l.length;i++){const ch=l[i];if(ch==='"'){if(inQ&&l[i+1]==='"'){c+='"';i++;}else inQ=!inQ;}else if(ch===sep&&!inQ){f.push(c.trim());c='';}else c+=ch;}f.push(c.trim());return f;}
  if(lines.length<2)return{headers:[],rows:[]};
  const headers=pl(lines[0]);const rows=[];
  for(let i=1;i<lines.length;i++){const ln=lines[i].trim();if(!ln)continue;const vals=pl(ln);if(vals.every(v=>!v))continue;const obj={};headers.forEach((h,idx)=>{obj[h]=vals[idx]||'';});rows.push(obj);}
  return{headers,rows};
}

}

/* ── Cascade availability ── */
function matchExcept(rv,xid){
  for(const id of FILTER_IDS){
    if(id===xid)continue;
    const vs=FC[id]?.values;
    if(!vs||vs.size===0)continue;
    if(!vs.has(rv[FMAP[id]]||''))return false;
  }
  return true;
}
function computeAvail(){
  const av={};
  for(const id of FILTER_IDS){
    const s=new Set();
    for(const r of allRows){const rv=getRowVals(r);if(matchExcept(rv,id)&&rv[FMAP[id]])s.add(rv[FMAP[id]]);}
    av[id]=s;
  }
  return av;
}

/* ── Fórmula real por pneu ── */
function buildFormulaTxt(nacImport, recapes, mm){
  const rec = isNaN(recapes) ? 0 : Math.max(0, Math.min(3, Math.round(recapes)));
  const rpmm = rec === 0 ? 90 : 55;

  const ni = norm(nacImport||'');
  const tipo = ni.includes('nac') ? 'NACIONAL' : ni.includes('imp') ? 'IMPORTADO' : null;

  const carcaca = tipo && carcacaMap[tipo] ? (carcacaMap[tipo][rec] ?? null) : null;

  const mmVal = isNaN(mm) || mm <= 0 ? null : mm;

  if(carcaca === null && mmVal === null){
    return `R$${rpmm}/mm`;
  }

  const carcacaStr = carcaca !== null ? fmtBRL(carcaca) : '?';
  const mmStr      = mmVal   !== null ? mmVal            : '?';
  const rpmmStr    = `R$${rpmm}/mm`;

  if(carcaca !== null && mmVal !== null){
    const total = carcaca + (mmVal * rpmm);
    return `${carcacaStr} + (${mmStr}mm × ${rpmmStr}) = ${fmtBRL(total)}`;
  }
  return `${carcacaStr} + (${mmStr}mm × ${rpmmStr})`;
}

  }

/* ── Load ── */
async function loadData(){
  showOv('⏳','Carregando dados…','Aguarde enquanto buscamos a planilha.');
  try{
    const res=await fetch(`${CSV_URL}&_ts=${Date.now()}`,{cache:'no-store'});
    if(!res.ok)throw new Error(`HTTP ${res.status}`);
    const buf=await res.arrayBuffer();
    const text=new TextDecoder('utf-8').decode(buf);
    const{headers,rows}=parseCSV(text);
    if(!rows.length)throw new Error('Nenhuma linha encontrada no CSV.');

    keys={
      data:       findKey(headers,['DATA COLETA','DATA']),
      fornecedor: findKey(headers,['FORNECEDOR']),
      fogo:       findKey(headers,['FOGO PNEU','FOGO']),
      marca:      findKey(headers,['MARCA PNEU','MARCA']),
      modelo:     findKey(headers,['MODELO PNEU','MODELO']),
      banda:      findKey(headers,['BANDA']),
      dot:        findKey(headers,['DOT']),
      qtdeRecape: findKey(headers,['QTDE RECAPE']),
      mm:         findKey(headers,['M.M','MM','QTD MM','MILIMETROS']),
      nacImport:  findKey(headers,['NAC / IMPORT','NAC/IMPORT','NACIMPORT']),
      servico:    findKey(headers,['SERVIÇO','SERVICO']),
      acao:       findKey(headers,['AÇÃO','ACAO']),
      kmRodado:   findKey(headers,['KM RODADO PNEU','KM RODADO']),
      recusado:   findKey(headers,['RECUSADO']),
      dataColeta: findKey(headers,['DATA']),
      valor:      findKey(headers,['VALOR']),
      motivoDescarte: findKey(headers,['Motivos de Descarte','MOTIVOS DESCAR']),
      obs:        findKey(headers,['OBS']),
    };

    console.log('Headers:', headers);
    console.table(Object.fromEntries(Object.entries(keys).map(([k,v])=>[k,v||'❌'])));

    const missing=Object.entries(keys).filter(([k,v])=>!v&&['data','marca'].includes(k));
    if(missing.length){
      const dp=document.getElementById('debug');dp.style.display='block';
      dp.innerHTML=`⚠️ Colunas não mapeadas: ${missing.map(([k])=>k).join(', ')} | Colunas: ${headers.map(h=>`"${h}"`).join(' · ')}`;
    }else{document.getElementById('debug').style.display='none';}

    const currentYear=new Date().getFullYear();
    allRows=rows.filter(r=>{
      const acao=norm(r[keys.acao]||'');
      const servico=norm(r[keys.servico]||'');
      const isSucataOrTB=acao.includes('sucata')||acao.includes('tb')||servico.includes('sucata')||servico.includes('tb');
      if(!isSucataOrTB)return false;
      const d=toDate(r[keys.data]);
      if(!d||d.getFullYear()!==currentYear)return false;
      return r[keys.marca]?.trim()||r[keys.data]?.trim();
    });

    initFilters(allRows);
    document.getElementById('filter-count').textContent=`${allRows.length} registros`;
    await loadCarcacas();
    hideOv();renderAll(allRows);
    document.getElementById('last-update').textContent='Atualizado em: '+new Date().toLocaleString('pt-BR');
  }catch(err){
    showOv('❌','Erro ao carregar',`${err.message} — Abra o console (F12) para mais detalhes.`);
    console.error(err);
  }
}

  const track=document.getElementById('tbl-scrollbar-track');
  if(!wrap||!thumb||!track)return;

  function updateThumb(){
    const ratio=wrap.scrollWidth>wrap.clientWidth?wrap.clientWidth/wrap.scrollWidth:1;
    const thumbW=Math.max(ratio*track.clientWidth,40);
    const maxLeft=track.clientWidth-thumbW;
    const scrollRatio=wrap.scrollLeft/(wrap.scrollWidth-wrap.clientWidth)||0;
    thumb.style.width=thumbW+'px';
    thumb.style.left=(scrollRatio*maxLeft)+'px';
    thumb.style.opacity=ratio<1?'1':'0';
  }
  wrap.addEventListener('scroll',updateThumb);
  window.addEventListener('resize',updateThumb);
  window.addEventListener('tbl-rendered',updateThumb);

  let dragging=false,startX=0,startScroll=0;
  thumb.addEventListener('mousedown',e=>{dragging=true;startX=e.clientX;startScroll=wrap.scrollLeft;thumb.style.cursor='grabbing';e.preventDefault();});
  document.addEventListener('mousemove',e=>{if(!dragging)return;const thumbW=thumb.offsetWidth;const maxLeft=track.clientWidth-thumbW;const dx=e.clientX-startX;const ratio=dx/maxLeft;wrap.scrollLeft=startScroll+ratio*(wrap.scrollWidth-wrap.clientWidth);});
  document.addEventListener('mouseup',()=>{dragging=false;thumb.style.cursor='grab';});
  track.addEventListener('click',e=>{if(e.target===thumb)return;const rect=track.getBoundingClientRect();const clickX=e.clientX-rect.left;wrap.scrollLeft=(clickX/track.clientWidth)*(wrap.scrollWidth-wrap.clientWidth);});
  setTimeout(updateThumb,500);
});

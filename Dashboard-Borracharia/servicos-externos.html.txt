function parseCSV(raw){
  let lines=raw.split(/\r?\n/);
  let sep=','; if(lines[0] && lines[0].includes(';')) sep=';';
  function pl(l){ let f=[], c='', inQ=false; for(let i=0;i<l.length;i++){ let ch=l[i]; if(ch==='"'){ if(inQ && l[i+1]==='"'){ c+='"'; i++; } else inQ=!inQ; } else if(ch===sep && !inQ){ f.push(c.trim()); c=''; } else c+=ch; } f.push(c.trim()); return f; }
  if(lines.length<2) return {headers:[], rows:[]};
  let headers=pl(lines[0]);
  let rows=[];
  for(let i=1;i<lines.length;i++){ let ln=lines[i].trim(); if(!ln) continue; let vals=pl(ln); if(vals.every(v=>!v)) continue; let obj={}; headers.forEach((h,idx)=>{ obj[h]=vals[idx]||''; }); rows.push(obj); }
  return {headers, rows};
}

function renderLinha(rows){
  let map = {};
  rows.forEach(r => {
    let d = toDate(r[keys.data]);
    if (!d || isNaN(d)) return;
    let mm = String(d.getMonth() + 1).padStart(2, '0');
    let lbl = `${MESES_ABREV[mm]}/${d.getFullYear()}`;
    let v = parseMoney(r[keys.valor]);
    if (v > 0) map[lbl] = (map[lbl] || 0) + v;
  });
  
  let labels = Object.keys(map).sort((a, b) => {
    let [ma, ya] = a.split('/');
    let [mb, yb] = b.split('/');
    let mesA = MESES_ORDEM.indexOf(ma);
    let mesB = MESES_ORDEM.indexOf(mb);
    if (ya !== yb) return parseInt(ya) - parseInt(yb);
    return mesA - mesB;
  });
  
  chartLinha = destroyChart(chartLinha);
  let ctx = document.getElementById('chart-linha').getContext('2d');
  let grad = ctx.createLinearGradient(0, 0, 0, 200);
  grad.addColorStop(0, 'rgba(26,92,255,.15)');

  chartLinha = new Chart(ctx, {
    type: 'line',

      datasets: [{

        data: labels.map(l => map[l]),
        borderColor: '#1A5CFF',
        backgroundColor: grad,
        borderWidth: 2.5,
        pointRadius: 5,
        pointHoverRadius: 7,
        pointBackgroundColor: '#fff',
        pointBorderColor: '#1A5CFF',
        pointBorderWidth: 2,
        tension: .35,
        fill: true
      }]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      onClick: (e, els) => {
        if (!els.length) return;
        let lbl = labels[els[0].index];
        let [mm, yyyy] = lbl.split('/');
        let mNum = Object.keys(MESES_ABREV).find(k => MESES_ABREV[k] === mm);
        if (mNum) {
          fcToggle('fc-ano', yyyy);
          fcToggle('fc-mes', mNum);
        }
      },
      plugins: {
        legend: { display: false },
        tooltip: {
          callbacks: {
            label: c => fmtBRL(c.parsed.y)
          }
        }
      },
      scales: {
        x: {
          grid: { color: '#E2E8F0' },
          ticks: { font: { family: 'Nunito', size: 11 }, color: '#64748B' }
        },

async function loadData(){
  showMsg('⏳','Carregando dados…','Buscando informações da planilha. Aguarde.');
  try{
    let res=await fetch(`${CSV_URL}&_ts=${Date.now()}`,{cache:'no-store'});
    if(!res.ok) throw new Error(`HTTP ${res.status}`);
    let text=await res.text();
    let {headers,rows}=parseCSV(text);
    if(!rows.length) throw new Error('Nenhuma linha encontrada.');
    keys={
      data: findKey(headers,['DATA']),
      placa: findKey(headers,['PLACA']),
      valor: findKey(headers,['VALOR']),
      motorista: findKey(headers,['MOTORISTA']),
      cliente: findKey(headers,['CLIENTE']),
      servico: findKey(headers,['SERVIÇO']),
      segmento: findKey(headers,['SEGMENTO','SEGUIMENTO','TIPO DE VEÍCULO','TIPO VEICULO','TIPO']),
      solicitante: findKey(headers,['SOLICITANTE'])
    };
    let falta=[];
    if(!keys.data) falta.push('DATA');
    if(!keys.valor) falta.push('VALOR');
    if(!keys.cliente) falta.push('CLIENTE');
    if(!keys.motorista) falta.push('MOTORISTA');
    if(falta.length){ showMsg('⚠️','Colunas não encontradas',`Campos faltando: ${falta.join(', ')}\n\nColunas lidas:\n${headers.join(' | ')}`); return; }
    allRows=rows.filter(r=>r[keys.data]?.trim());
    initFilters(allRows);
    document.getElementById('filter-count').textContent=`${allRows.length} registros`;
    hideMsg();
    renderAll(allRows);
    document.getElementById('last-update').textContent='Atualizado em: '+new Date().toLocaleString('pt-BR');
  } catch(err){ showMsg('❌','Erro ao carregar',err.message); console.error(err); }
}

document.getElementById('btn-refresh').onclick=loadData;
document.getElementById('btn-limpar').onclick=clearFilters;
loadData();

function norm(s){return String(s||'').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/[\/\\]+/g,'').replace(/[\s_\-\.]+/g,'');}

function findKey(headers,target){
  // Tenta match exato primeiro, depois match normalizado
  const t=norm(target);
  return headers.find(h=>h===target) || 
         headers.find(h=>norm(h)===t) || 
         headers.find(h=>norm(h).includes(t)||t.includes(norm(h))) || 
         null;
}

// FUNÇÃO CORRIGIDA: Trata especificamente o formato "DD de MÊS de AAAA"
function toDate(v){
  if(!v||!String(v).trim()) return null;
  let s=String(v).trim();
  
  console.log('Tentando converter data:', s);
  
  // Formato "DD de MÊS de AAAA" ou "DD de MÊS. de AAAA"
  const mPt = s.match(/^(\d{1,2})\s+de\s+([a-zç]+)\.?\s+de\s+(\d{2,4})$/i);
  if(mPt){
    const day=parseInt(mPt[1],10);
    const mesNome=mPt[2].toLowerCase().replace(/\./g,'');
    const mesNum=MESES_PT[mesNome];
    const y=mPt[3].length===2?2000+parseInt(mPt[3],10):parseInt(mPt[3],10);
    
    if(mesNum){
      const d=new Date(y, mesNum-1, day);
      if(!isNaN(d.getTime())){
        console.log('Data convertida (formato PT):', d);
        return d;
      }
    }
  }
  
  // Formato DD/MM/YYYY ou DD-MM-YYYY
  const mSlash = s.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{2,4})$/);
  if(mSlash){
    const day=parseInt(mSlash[1],10), mon=parseInt(mSlash[2],10);
    const y=mSlash[3].length===2?2000+parseInt(mSlash[3],10):parseInt(mSlash[3],10);
    const d=new Date(y, mon-1, day);
    if(!isNaN(d.getTime())) {
      console.log('Data convertida (formato BR):', d);
      return d;
    }
  }
  
  // Formato YYYY-MM-DD (ISO)
  const mIso = s.match(/^(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})$/);
  if(mIso){
    const y=parseInt(mIso[1],10), mon=parseInt(mIso[2],10), day=parseInt(mIso[3],10);
    const d=new Date(y, mon-1, day);
    if(!isNaN(d.getTime())) {
      console.log('Data convertida (formato ISO):', d);
      return d;
    }
  }
  
  // Última tentativa: new Date() direto
  const d=new Date(s);
  if(!isNaN(d.getTime())){
    console.log('Data convertida (new Date):', d);
    return d;
  }
  
  console.log('Falha ao converter data:', s);
  return null;
}

function parseBool(v){
  const n=norm(v||'');
  if(n==='sim'||n==='s'||n==='ok'||n==='x') return 'SIM';
  if(n==='nao'||n==='não'||n==='no'||n==='n'||n==='não ok'||n==='nãook'||n==='naook') return 'NÃO';
  if(n===''||n==='-'||n==='nd'||n==='na'||n==='naoaplicavel') return 'N.D.';
  return 'NÃO';
}

    case 'fc-ano':        { const d=toDate(r[keys.data]); return d?String(d.getFullYear()):null; }
    case 'fc-mes':        { const d=toDate(r[keys.data]); return d?String(d.getMonth()+1).padStart(2,'0'):null; }
    case 'fc-calibragem': return parseBool(r[keys.calibragem]);
    case 'fc-reaperto':   return parseBool(r[keys.reaperto]);
    case 'fc-mapeamento': return parseBool(r[keys.mapeamento]);
    default: return null;
  }
}

function getRowsExcluding(excludeId){
  return allRows.filter(r=>{
    for(const id of ALL_FILTER_IDS){
      if(id===excludeId) continue;
      const val=FC[id]?.value; if(!val) continue;
      if(getRowFieldValue(r,id)!==val) return false;
    }
    return true;
  });
}

function getFilteredRows(){
  return allRows.filter(r=>{
    for(const id of ALL_FILTER_IDS){
      const val=FC[id]?.value; if(!val) continue;
      if(getRowFieldValue(r,id)!==val) return false;
    }
    return true;
  });
}

function updateAvailableOptions(){
  for(const id of ALL_FILTER_IDS){
    if(!FC[id]) continue;
    const rows=getRowsExcluding(id);
    const avail=new Set();
    rows.forEach(r=>{const v=getRowFieldValue(r,id);if(v) avail.add(v);});
    FC[id].availableOptions=FC[id].allOptions.filter(o=>avail.has(o));
  }
}

function fcSetValue(wrapId, value, label){
  const wrap=document.getElementById(wrapId); if(!wrap) return;
  const input=wrap.querySelector('.fc-input');
  const dropdown=wrap.querySelector('.fc-dropdown');
  if(!FC[wrapId]) return;
  if(value&&!FC[wrapId].availableOptions.includes(value)){showToast('⚠️ Sem registros com os filtros atuais');return;}
  const prev=FC[wrapId].value;
  if(prev===value&&value!==''){
    FC[wrapId].value=''; input.value=''; input.classList.remove('has-value'); showToast('Filtro removido');
  } else {
    FC[wrapId].value=value;
    const displayLabel=wrapId==='fc-mes'&&value?(MESES_NOME[value]||label||value):(label||value);
    input.value=displayLabel; input.classList.toggle('has-value',value!=='');
    if(value) showToast(`Filtro: ${displayLabel}`);
  }
  if(dropdown) dropdown.classList.remove('open');
  applyFilters();
}

function parseCSVRobust(raw){
  const lines=raw.split(/\r?\n/);
  
  function parseLine(line){
    const fields=[];
    let cur='',inQ=false;
    
    for(let i=0;i<line.length;i++){
      const ch=line[i];
      if(ch==='"'){
        if(inQ&&line[i+1]==='"'){
          cur+='"';
          i++;
        }else{
          inQ=!inQ;
        }
      }
      else if(ch===','&&!inQ){
        fields.push(cur.trim());
        cur='';
      }
      else{
        cur+=ch;
      }
    }
    fields.push(cur.trim());
    return fields;
  }
  
  if(lines.length<2){
    console.error('CSV muito curto');
    return {headers:[],rows:[]};
  }
  
  const headers=parseLine(lines[0]);
  console.log('Headers encontrados:', headers);
  
  const rows=[];
  for(let i=1;i<lines.length;i++){
    const ln=lines[i].trim();
    if(!ln) continue;
    
    const vals=parseLine(ln);
    
    if(vals.every(v=>!v)) continue;
    
    const obj={};
    headers.forEach((h,idx)=>{
      obj[h]=vals[idx]||'';
    });
    rows.push(obj);
  }
  
  console.log(`Total de ${rows.length} linhas de dados`);
  if(rows.length>0) console.log('Primeira linha:', rows[0]);
  
  return {headers,rows};
}

async function loadData(){
  showMsg('⏳','Carregando dados…','Buscando informações da planilha. Aguarde.');
  
  try{
    const res=await fetch(`${CSV_URL}&_ts=${Date.now()}`,{cache:'no-store'});
    if(!res.ok) throw new Error(`HTTP ${res.status}`);
    
    const text = await res.text();
    console.log('CSV recebido (primeiros 500 caracteres):', text.substring(0, 500));
    
    const {headers,rows}=parseCSVRobust(text);
    
    if(!rows.length) {
      console.error('Headers:', headers);
      throw new Error('Nenhuma linha de dados encontrada.');
    }
    
    console.log('Primeira linha:', rows[0]);
    console.log('Headers:', headers);
    
    // Mapeamento EXATO com os nomes da sua planilha
    keys={
      data:        findKey(headers,'DATA'),
      placa:       findKey(headers,'PLACAEQUIPAMENTO'),
      tipo:        findKey(headers,'CAVALO / CARRETA'),
      motorista:   findKey(headers,'NOME DO MOTORISTA'),
      calibragem:  findKey(headers,'CALIBRAGEM DOS PNEUS'),
      reaperto:    findKey(headers,'REAPERTO RODAS'),
      mapeamento:  findKey(headers,'MAPEAMENTO PNEUS'),
    };
    
    console.log('Chaves mapeadas:', keys);
    
    const falta=[];
    if(!keys.data)       falta.push('DATA');
    if(!keys.placa)      falta.push('PLACAEQUIPAMENTO');
    if(!keys.tipo)       falta.push('CAVALO / CARRETA');
    if(!keys.motorista)  falta.push('NOME DO MOTORISTA');
    if(!keys.calibragem) falta.push('CALIBRAGEM DOS PNEUS');
    if(!keys.reaperto)   falta.push('REAPERTO RODAS');
    if(!keys.mapeamento) falta.push('MAPEAMENTO PNEUS');
    
    if(falta.length){
      showMsg('⚠️','Colunas não encontradas',
        `Campos faltando: ${falta.join(', ')}\n\nColunas encontradas: ${headers.join(' | ')}`);
      console.error('Colunas faltando:', falta);
      return;
    }
    
    allRows=rows.filter(r=>r[keys.placa]?.trim()||r[keys.data]?.trim());
    console.log('Total de linhas:', allRows.length);
    
    // Testa conversão de data na primeira linha
    if(allRows.length>0){
      console.log('Teste data primeira linha:', allRows[0][keys.data]);
      console.log('Resultado:', toDate(allRows[0][keys.data]));
    }
    
    initFilters(allRows);
    document.getElementById('filter-count').textContent=`${allRows.length} registros`;
    
    hideMsg();
    renderAll(allRows);
    document.getElementById('last-update').textContent='Atualizado em: '+new Date().toLocaleString('pt-BR');
    
  }catch(err){
    showMsg('❌','Não foi possível carregar',`Erro: ${err.message}`);
    console.error('Erro:', err);
  }
}

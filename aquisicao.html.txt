    getRowsExcluding(id).forEach(r=>{const v=getRowFieldValue(r,id);if(v)s.add(v);});
    av[id]=s;
  }
  return av;
}

/* ── Render dropdown ── */
function fcRenderDD(wid,q){
  const wrap=document.getElementById(wid);if(!wrap)return;
  const dd=wrap.querySelector('.fc-dropdown');
  const st=FC[wid];if(!st)return;
  const avSet=computeAvail()[wid];
  const query=(q||'').toLowerCase().trim();
  const fq=l=>query?l.filter(o=>getLabel(wid,o).toLowerCase().includes(query)):l;
  const av=fq(st.allOptions.filter(o=>avSet.has(o)));
  const unav=fq(st.allOptions.filter(o=>!avSet.has(o)));
  dd.innerHTML='';
  function mk(opt,isU){
    const div=document.createElement('div');
    const sel=st.values.has(opt);
    div.className='fc-option'+(sel?' selected':'')+(isU?' unavail':'');
    const chk=document.createElement('span');chk.className='fc-chk';
    const lbl=document.createElement('span');
    if(query&&!isU){const s=getLabel(wid,opt);const idx=s.toLowerCase().indexOf(query);if(idx>=0){lbl.innerHTML=s.slice(0,idx)+'<em style="font-style:normal;font-weight:700;color:var(--accent)">'+s.slice(idx,idx+query.length)+'</em>'+s.slice(idx+query.length);}else lbl.textContent=s;}

    div.appendChild(chk);div.appendChild(lbl);
    if(!isU)div.addEventListener('mousedown',e=>{e.preventDefault();fcToggle(wid,opt);});

  }

    av.forEach(o=>dd.appendChild(mk(o,false)));
  }

    const l=document.createElement('div');l.className='fc-section-label';l.textContent='Sem resultado nos filtros atuais';dd.appendChild(l);
    unav.forEach(o=>dd.appendChild(mk(o,true)));
  }

}

/* ── Toggle ── */
function fcToggle(wid,val){
  const st=FC[wid];if(!st)return;
  if(st.values.has(val)){st.values.delete(val);showToast(`✕ ${getLabel(wid,val)}`);}
  else{st.values.add(val);showToast(`✓ ${getLabel(wid,val)}`);}
  fcSyncBox(wid);applyFilters();
  const wrap=document.getElementById(wid);
  const txt=wrap?.querySelector('.fc-text');
  fcRenderDD(wid,txt?txt.value:'');
}

/* ── CSV parser ── */
function parseCSVRobust(raw){
  const lines=raw.split(/\r?\n/);
  function parseLine(line){const fields=[];let cur='',inQ=false;for(let i=0;i<line.length;i++){const ch=line[i];if(ch==='"'){if(inQ&&line[i+1]==='"'){cur+='"';i++;}else{inQ=!inQ;}}else if(ch===','&&!inQ){fields.push(cur.trim());cur='';}else{cur+=ch;}}fields.push(cur.trim());return fields;}
  if(lines.length<2)return{headers:[],rows:[]};
  const headers=parseLine(lines[0]);const rows=[];
  for(let i=1;i<lines.length;i++){const ln=lines[i].trim();if(!ln)continue;const vals=parseLine(ln);if(vals.every(v=>!v))continue;const obj={};headers.forEach((h,idx)=>{obj[h]=vals[idx]||'';});
  // Ignora linhas que contenham exatamente "CIA VERDE - AJUSTE SISTEMA"
  const rowStr=Object.values(obj).map(v=>(v||'').trim()).join('|');
  if(rowStr.includes('CIA VERDE - AJUSTE SISTEMA'))continue;
  rows.push(obj);}
  return{headers,rows};
}

  });

  document.getElementById('kpi-valor-total').textContent=fmtBRL(somaNF);
  document.getElementById('kpi-valor-medio').textContent=fmtBRL(cNF>0?somaNF/cNF:0);
  document.getElementById('kpi-novos').textContent=novos.toLocaleString('pt-BR');
  document.getElementById('kpi-novos-pct').textContent=total?`${((novos/total)*100).toFixed(1)}% do total`:'';

  document.getElementById('kpi-recapados-pct').textContent=total?`${((recapados/total)*100).toFixed(1)}% do total`:'';
}

/* ── Load ── */
async function loadData(){
  showMsg('⏳','Carregando dados…','Buscando informações da planilha. Aguarde.');
  try{
    const res=await fetch(`${CSV_URL}&_ts=${Date.now()}`,{cache:'no-store'});
    if(!res.ok)throw new Error(`HTTP ${res.status}`);
    const{headers,rows}=parseCSVRobust(await res.text());
    if(!rows.length)throw new Error('Nenhuma linha de dados encontrada.');
    keys={
      data:       findKey(headers,'DATA'),
      marca:      findKey(headers,'MARCA'),
      fornecedor: findKey(headers,'FORNECEDOR'),
      espec:      findKey(headers,'ESPECIFICACAO'),
      status:     findKey(headers,'NOVO  RECAPADO'),
      vlUnit:     findKey(headers,'VALOR UNITAR'),
      vlNF:       findKey(headers,'VALOR TOTAL NF'),
      banda:      findKey(headers,'BANDA'),
    };
    const falta=[];
    if(!keys.data)falta.push('DATA');
    if(!keys.marca)falta.push('MARCA');
    if(!keys.status)falta.push('NOVO / RECAPADO');
    if(!keys.vlUnit&&!keys.vlNF)falta.push('VALOR UNITÁR / VALOR TOTAL NF');
    if(falta.length){showMsg('⚠️','Colunas não encontradas',`Campos: ${falta.join(', ')}\n\nColunas lidas: ${headers.join(' | ')}`);return;}
    allRows=rows.filter(r=>r[keys.marca]?.trim()||r[keys.data]?.trim());
    initFilters(allRows);
    document.getElementById('filter-count').textContent=`${allRows.length} registros`;
    hideMsg();
    renderAll(allRows);
    document.getElementById('last-update').textContent='Atualizado em: '+new Date().toLocaleString('pt-BR');
  }catch(err){
    showMsg('❌','Não foi possível carregar',`Erro: ${err.message}`);
    console.error(err);
  }
}
